### LeetCode No.18
- 思路同三数之和，多出来的数**k**改变了一些处理:
	- target可能是负数，所以 `k和i` 的剪枝的条件是`nums[k]>=0和nums[i]>=0` 的情况下
	- 去重:nums[k] == nums[k -1];

```java

```class Solution {
    public List<List<Integer>> fourSum(int[] nums, int target) {
        List<List<Integer>> result = new ArrayList<>();
        Arrays.sort(nums);

        for(int k = 0;k < nums.length;k++){
            if(nums[k] > target && nums[k] >= 0){
                return result;//>=0的情况下
            }
            
            if(k > 0 && nums[k] == nums[k -1]){
                continue;
            }

            for(int i = k + 1;i < nums.length;i++){
                if(nums[i] + nums[k] > target && nums[i] + nums[k] >= 0){
                    break;//此时return result会有遗漏
                }
                if(i > k  + 1 && nums[i] == nums[i - 1]){
                    continue;
                }

                int left = i+1;
                int right = nums.length - 1;
               
                while(left < right){ 
                    int sum = nums[k] + nums[i] + nums[left] + nums[right];//必须在循环内，才能让nums[left/right]改变
                    if(sum > target){
                        right--;
                    }
                    else if(sum < target){
                        left++;
                    }
                    else{
                        result.add(Arrays.asList(nums[k],nums[i],nums[left],nums[right]));

                        while(right > left && nums[right] == nums[right - 1]) right--;//条件right > left,防止{0,0,0,0}的情况，死循环，所以从两边开始去重
                        while(right > left && nums[left] == nums[left + 1]) left++;
                        right--;
                        left++;
                    }   
                }
            }
        }
        return result;   
    }
}
