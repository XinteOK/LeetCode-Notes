- 采用创建新数组填充的方法

```java
Class Solution {
	public String reverseWords(String s){
		//转换为字符数组
		char[] initialArr = s.toCharArray();
		//创建被填充的新数组
		char[] newArr = new char[initialArr.length + 1];//要+1 
		//创建新数组的指针
		int newArrPos = 0;
		//填充
		//创建读指针
		int read = initialArr.length - 1;//从最后开始读
		while(read>=0){//读到开头
			//跳过所有“空格”，填充时单独加“空格”
			while(read>=0 && initialArr[read]==' '){read--;}
			//读单词
			//记录单词右边界
			int right = read;
			while(read>=0 && initialArr[read]!=' '){read--;}
			//填充
			for(int write = read + 1;write<=right;write++){
				newArr[newArrPos++] = initialArr[write];
				//每个单词后面单独加一个“空格”
				if(write == right){
					newArr[newArrPos++] = ' ';
				}
			}
		}
		//返回答案
		if(newArrPos == 0){
			return "";//""表示空字符串
		}
		else return new String(newArr,0,newArrPos - 1);//返回除去最后一个单词的多余空格的新数组转换的的字符串
	} 
}
```

```java
new String(字符数组,起始下标,字符数量);
```
